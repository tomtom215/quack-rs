# The SQL checks in README.md's "Load in DuckDB CLI" block, with the exact
# output the DuckDB CLI prints for each (`-csv -noheader -nullvalue NULL`).
# scripts/check-hello-ext.py runs them against a loaded extension and fails
# if README.md's statements and these ever differ.
#
# Format: `== <id>`, the statement, `----`, the expected output (one line per
# row, columns comma-separated; `ERROR: <prefix>` for an expected error).

== T01
SELECT word_count(sentence) AS wc FROM ( VALUES ('hello world'), ('one two three'), (NULL)) t(sentence);
----
5

== T02
SELECT first_word('hello world');
----
hello

== T03
SELECT COUNT(*) FROM generate_series_ext(5);
----
5

== T04
SELECT COUNT(*) FROM generate_series_ext(0);
----
0

== T05
SELECT CAST('3.14' AS INTEGER);
----
ERROR: Conversion Error

== T06
SELECT TRY_CAST('1e5' AS INTEGER);
----
NULL

== T07
SELECT sum_list([1, 2, 3]);
----
6

== T08

----

== T09
SELECT make_pair('hello', 42);

{'key': hello, 'value': 42}
